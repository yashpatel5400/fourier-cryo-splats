"""Conservative Gaussian calibration allowing unknown signal means.

The common covariance and independence assumptions are essential. These helpers
do not establish those assumptions for archive particles or fitted poses.
"""
import numpy as np
from scipy.optimize import brentq


def common_covariance_trace_upper(observations, failure_probability=.005):
    """Upper bound tr(Sigma) from independent N(mu_i,Sigma) vectors.

    For t in (0,1) solving n*(t-1-log(t))/2 = log(1/beta), the bound is
    sum_i ||Y_i||^2/(n*t). Unknown means need not be equal or zero. This follows
    from the noncentral Gaussian Laplace transform and prod(1+x_j)>=1+sum x_j.
    It implies Sigma <= bound*I, usually very conservatively in many dimensions.
    """
    y=np.asarray(observations,dtype=float)
    if y.ndim!=2 or min(y.shape)<1 or not np.isfinite(y).all():
        raise ValueError('Finite nonempty matrix of independent observations required')
    if not 0<failure_probability<1:
        raise ValueError('Failure probability must lie in (0,1)')
    n=y.shape[0];target=2*np.log(1/failure_probability)/n
    # Solve for log(t) to avoid loss of significance near zero.
    log_t=brentq(lambda x:np.expm1(x)-x-target,-max(2.,2*target+2),0.)
    t=float(np.exp(log_t));energy=float(np.sum(y*y))
    return {'covariance_trace_upper':energy/(n*t),'lower_tail_fraction':t,
            'observations':n,'dimension':y.shape[1],'observed_total_energy':energy,
            'failure_probability':failure_probability,
            'assumptions':'Independent Gaussian vectors with one common covariance and arbitrary fixed means; real arithmetic.'}


def estimator_variance_upper(observations, raw_weights, failure_probability=.005):
    """Bound noise variance of sum_i v_i.T Y_i from independent calibration.

    Each row of raw_weights is one inference particle's coefficient in the
    ORIGINAL calibration coordinate frame. The weights must be independent of
    these calibration observations, but may depend on separate training data.
    Applying the trace lemma to V Y_j gives tr(V Sigma V.T), exactly the sum
    of inference variances under independent particles with common Sigma.
    No independence between the projected coordinates is assumed.
    """
    y=np.asarray(observations,dtype=float);v=np.asarray(raw_weights,dtype=float)
    if y.ndim!=2 or v.ndim!=2 or y.shape[1]!=v.shape[1] or min(v.shape)<1 or not np.isfinite(v).all():
        raise ValueError('Finite compatible calibration observations and raw-frame weight matrix required')
    result=common_covariance_trace_upper(y@v.T,failure_probability)
    return {'estimator_variance_upper':result['covariance_trace_upper'],
            'noise_sd_upper':float(np.sqrt(result['covariance_trace_upper'])),
            'calibration':result,'inference_particles':len(v),
            'raw_coordinate_dimension':v.shape[1],
            'assumptions':'Weights independent of this calibration pool; independent Gaussian calibration and inference particles with a common covariance in the raw coordinate frame; arbitrary fixed means.'}


def uncenter_fourier_weights(weights, phases, whitening_scale=1.):
    """Pull realified weights back through complex unit-modulus image phases."""
    p=np.asarray(phases,dtype=complex);w=np.asarray(weights,dtype=float)
    if p.ndim!=2 or w.shape!=(p.shape[0],2*p.shape[1]) or not np.isfinite(w).all() or not np.isfinite(p).all():
        raise ValueError('Compatible finite particle-wise weights and phases required')
    if not np.allclose(abs(p),1.,rtol=1e-12,atol=1e-12) or not np.isfinite(whitening_scale) or whitening_scale<=0:
        raise ValueError('Unit-modulus phases and positive whitening scale required')
    nq=p.shape[1];raw=(w[:,:nq]+1j*w[:,nq:])*p.conjugate()/whitening_scale
    return np.concatenate([raw.real,raw.imag],axis=1)
